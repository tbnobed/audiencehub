BEGIN READ ONLY;
SET LOCAL statement_timeout = '60s';

SELECT id, type, status, heartbeat_at, progress
FROM jobs
WHERE type IN ('import.run', 'identity.resolve_batch', 'traits.recompute')
ORDER BY id DESC LIMIT 15;

SELECT count(*) AS unresolved_records
FROM source_records
WHERE resolved_at IS NULL;

SELECT count(*) AS gifts,
       count(*) FILTER (WHERE profile_id IS NULL) AS unlinked_gifts,
       min(gift_date) AS earliest_gift,
       max(gift_date) AS latest_gift
FROM gifts;

COMMIT;
